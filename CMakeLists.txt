cmake_minimum_required(VERSION 3.16)
project(UMDI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(umdi_core src/virtual_biossd.cpp src/device_controller.cpp)
target_include_directories(umdi_core PUBLIC include)
add_executable(umdi_demo src/main.cpp)
target_link_libraries(umdi_demo PRIVATE umdi_core)
add_executable(umdi_tests tests/roundtrip.cpp)
target_link_libraries(umdi_tests PRIVATE umdi_core)
enable_testing()
add_test(NAME roundtrip COMMAND umdi_tests)
